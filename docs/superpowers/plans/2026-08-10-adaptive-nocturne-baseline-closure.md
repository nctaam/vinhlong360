export const WINDOWS_EXACT_PROCESS_CLEANUP_TIMEOUT_MS = 30_000
export const CONTROL_HELPER_JOB_CLOSE_GRACE_MS = 2_000

  return platform === 'win32' ? WINDOWS_EXACT_PROCESS_CLEANUP_TIMEOUT_MS : 12_000

}
